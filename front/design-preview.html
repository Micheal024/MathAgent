import numpy as np
from scipy.optimize import linprog

# 从 df_supply 动态计算效益系数（修正后）
c_agri = df_supply['yield_per_unit'].mean()      # 农业边际效益
c_ind  = df_supply['industrial_gdp'].mean()      # 工业边际效益
c_life = df_supply['livability_score'].mean()    # 生活效益评分
c = [-c_agri, -c_ind, -c_life]                  # 最小化负效益

# 约束矩阵
A_ub = [
    [1, 1, 1],          # 总量约束: x1+x2+x3 <= W_total
    [-1, 0, 0],         # 农业下限: x1 >= 0.4 * W_total
]
b_ub = [W_total, -0.4 * W_total]
bounds = [(0, None)] * 3

result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
print(f"求解状态: {result.message}")
print(f"最优分配: 农业={result.x[0]:.1f} 工业={result.x[1]:.1f} 生活={result.x[2]:.1f}")
print(f"总效益值: {-result.fun:.4f}")
